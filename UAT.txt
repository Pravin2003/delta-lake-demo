def create_status_emoji(status):
    """Return emoji for given status"""
    emoji_map = {
        'PASS': '✅',
        'WARN': '⚠️',
        'FAIL': '❌',
        'ERROR': '🔥',
        'SKIPPED': '⏭️'
    }
    return emoji_map.get(status, '❓')

def print_test_header(test_num, test_name):
    """Print formatted test header"""
    print("\n" + "="*100)
    print(f"TEST {test_num}: {test_name}")
    print("="*100)

def get_mapped_column(view_name, col_name):
    """Get mapped column name for Databricks if exists, else return original"""
    if not col_name:
        return col_name
    if view_name in COLUMN_MAPPINGS and col_name in COLUMN_MAPPINGS[view_name]:
        mapped = COLUMN_MAPPINGS[view_name][col_name]
        if mapped is None or mapped == '':
            return None
        return mapped
    return col_name

def get_date_columns_list(view_name):
    """Get list of date/timestamp columns for a view"""
    configured_cols = DATE_COLUMNS.get(view_name)
    
    if configured_cols:
        if isinstance(configured_cols, str):
            return [configured_cols]
        elif isinstance(configured_cols, list):
            return configured_cols
        else:
            return []
    
    try:
        schema = spark.sql(f"DESCRIBE {REDSHIFT_CATALOG}.{REDSHIFT_SCHEMA}.{view_name}").toPandas()
        schema = schema[~schema['col_name'].str.startswith('#')]
        
        date_type_columns = []
        for _, row in schema.iterrows():
            col_name = row['col_name']
            data_type = row['data_type'].lower()
            
            if any(dt in data_type for dt in ['date', 'timestamp']):
                date_type_columns.append(col_name)
        
        return date_type_columns
    except:
        return []

def get_primary_date_column(view_name):
    """Get primary date column for filtering"""
    date_cols = get_date_columns_list(view_name)
    return date_cols[0] if date_cols else None

def validate_business_metrics(view_name, common_cols):
    """
    Validate that configured business metrics exist in common columns.
    Checks both original Redshift names and mapped Databricks names.
    Returns: (valid_metrics, invalid_metrics)
    """
    configured_metrics = BUSINESS_METRICS.get(view_name, [])
    
    if not configured_metrics:
        return [], []
    
    valid_metrics = []
    invalid_metrics = []
    
    for metric in configured_metrics:
        # Check if original Redshift column name exists in common_cols
        # OR if its mapped Databricks name exists in common_cols
        mapped_metric = get_mapped_column(view_name, metric)
        
        if metric in common_cols or (mapped_metric and mapped_metric in common_cols):
            valid_metrics.append(metric)
        else:
            invalid_metrics.append(metric)
    
    return valid_metrics, invalid_metrics
